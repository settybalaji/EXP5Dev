Application successfully deployed using Docker.

pipeline { agent any environment { DOCKER_IMAGE = 'your-dockerhub-username/jenkins-demo' DOCKER_CREDENTIALS = 'dockerhub-credentials' } stages { stage('Checkout') { steps { git branch: 'main', url: 'https://github.com/your-username/MyApplication.git' } } stage('Build Application') { steps { echo 'Building the application...' bat 'dir' } } stage('Build Docker Image') { steps { bat 'docker build -t %DOCKER_IMAGE%:latest .' } } stage('Push Docker Image') { steps { withCredentials([ usernamePassword( credentialsId: 'dockerhub-credentials', usernameVariable: 'DOCKER_USER', passwordVariable: 'DOCKER_PASS' ) ]) { bat 'docker login -u %DOCKER_USER% -p %DOCKER_PASS%' bat 'docker push %DOCKER_IMAGE%:latest' bat 'docker logout' } } } } post { success { echo 'Pipeline executed successfully!' } failure { echo 'Pipeline execution failed!' } } } FROM httpd:latest COPY index.html /usr/local/apache2/htdocs/ EXPOSE 80
